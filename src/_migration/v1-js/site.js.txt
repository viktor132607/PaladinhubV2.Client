window.updateCartBadge = window.updateCartBadge || function (count, animate) {
    var badge = document.getElementById('cart-badge');
    if (!badge) return;
    if (!count || count <= 0) {
        badge.classList.add('d-none');
        badge.textContent = '0';
        return;
    }
    badge.textContent = count;
    badge.classList.remove('d-none');
    if (animate) {
        badge.classList.add('cart-badge-pop');
        setTimeout(function () { badge.classList.remove('cart-badge-pop'); }, 300);
    }
};

window.fetchCartCount = window.fetchCartCount || async function () {
    async function tryEndpoint(url) {
        const r = await fetch(url, { credentials: 'same-origin' });
        if (!r.ok) throw 0;
        const d = await r.json();
        return typeof d === 'number' ? d : (d?.count ?? 0);
    }
    try {
        const n = await tryEndpoint('/Cart/CountJson');
        window.updateCartBadge(n, false);
    } catch {
        try {
            const n = await tryEndpoint('/Cart/Count');
            window.updateCartBadge(n, false);
        } catch {
            window.updateCartBadge(0, false);
        }
    }
};

let hoverTimer = null;

async function refreshMiniCart() {
    const panel = document.getElementById('mini-cart-panel');
    if (!panel) return;
    try {
        const res = await fetch('/Cart/Mini', { credentials: 'same-origin' });
        panel.innerHTML = await res.text();
    } catch {
        panel.innerHTML = '<div class="p-3 text-center text-muted">Couldn\'t load cart.</div>';
    }
}

async function showMiniCart() {
    const panel = document.getElementById('mini-cart-panel');
    if (!panel) return;
    panel.classList.remove('d-none');
    await refreshMiniCart();
}

function hideMiniCart() {
    const panel = document.getElementById('mini-cart-panel');
    if (!panel) return;
    panel.classList.add('d-none');
}

async function removeFromCart(productId) {
    const token = document.querySelector("input[name='__RequestVerificationToken']")?.value;
    const headers = Object.assign({ 'X-Requested-With': 'XMLHttpRequest' }, token ? { 'RequestVerificationToken': token } : {});
    try {
        const r1 = await fetch(`/Cart/RemoveProduct?id=${encodeURIComponent(productId)}`, { method: 'POST', credentials: 'same-origin', headers });
        if (r1.ok) return true;
    } catch { }
    try {
        const fd = new FormData();
        fd.append('productId', productId);
        if (token) fd.append('__RequestVerificationToken', token);
        const r2 = await fetch(`/Cart/Remove`, { method: 'POST', credentials: 'same-origin', headers, body: fd });
        if (!r2.ok) throw 0;
        const d = await r2.json().catch(() => ({ ok: true }));
        return d?.ok !== false;
    } catch { return false; }
}

document.addEventListener('click', async (e) => {
    const btn = e.target.closest('.mini-remove');
    if (!btn) return;
    e.preventDefault();
    const id = btn.dataset.id;
    if (!id) return;
    const ok = await removeFromCart(id);
    if (ok) {
        await refreshMiniCart();
        await window.fetchCartCount?.();
    }
});

document.addEventListener('cart:changed', async () => {
    try {
        await window.fetchCartCount?.();
        await refreshMiniCart?.();
    } catch { }
});

document.addEventListener('DOMContentLoaded', () => {
    const host = document.getElementById('nav-cart');
    if (host) {
        host.addEventListener('mouseenter', () => {
            clearTimeout(hoverTimer);
            showMiniCart();
        });
        host.addEventListener('mouseleave', () => {
            hoverTimer = setTimeout(hideMiniCart, 200);
        });
    }
    window.fetchCartCount && window.fetchCartCount();
    document.addEventListener('visibilitychange', () => {
        if (!document.hidden) {
            window.fetchCartCount && window.fetchCartCount();
        }
    });
    if (window.__cartJustAdded__ === true || window.__cartJustAdded__ === "true") {
        window.fetchCartCount().then(() => {
            const n = parseInt(document.getElementById('cart-badge')?.textContent || '0', 10);
            window.updateCartBadge(n, true);
        });
    }
    document.querySelectorAll(".alert-custom").forEach(el => {
        setTimeout(() => el.remove(), 3500);
    });
});
