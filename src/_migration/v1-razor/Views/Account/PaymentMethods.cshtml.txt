<script>
    (function () {
        const token = document.querySelector('#af-token input[name="__RequestVerificationToken"]')?.value || '';

        const on = (sel, evt, fn) => document.addEventListener(evt, e => {
            const t = e.target.closest(sel);
            if (t) fn(e, t);
        });

        on('.btn-set-default', 'click', async (e, btn) => {
            e.preventDefault();
            if (btn.classList.contains('disabled')) return;
            btn.classList.add('disabled');
            try {
                const id = btn.getAttribute('data-id');
                const body = new URLSearchParams({ id });
                await fetch('@Url.Action("SetDefaultPaymentMethod", "Account")', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/x-www-form-urlencoded; charset=UTF-8',
                        'RequestVerificationToken': token
                    },
                    body
                });
                location.reload();
            } catch {
                btn.classList.remove('disabled');
                alert('Failed to set default.');
            }
        });

        on('.btn-remove', 'click', async (e, btn) => {
            e.preventDefault();
            if (!confirm('Remove this payment method?')) return;
            btn.classList.add('disabled');
            try {
                const href = btn.getAttribute('data-href');
                await fetch(href, { method: 'GET', headers: { 'Accept': 'text/html' } });
                location.reload();
            } catch {
                btn.classList.remove('disabled');
                alert('Failed to remove.');
            }
        });
    })();
</script>
