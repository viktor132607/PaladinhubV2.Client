@section Scripts {
    <script>
        (function(){
            // prefs
            const K_HOVER="ph_enableMiniCartHover", K_NAV="ph_stickyNavbar";
            const chkHover=document.getElementById('prefMiniCartHover');
            const chkNav=document.getElementById('prefStickyNavbar');
            const btnReset=document.getElementById('btnResetPrefs');
            function getBool(k,d){const v=localStorage.getItem(k);return v===null?d:v==="true";}
            function setBool(k,val){localStorage.setItem(k,val?"true":"false");}
            if(chkHover) chkHover.checked=getBool(K_HOVER,true);
            if(chkNav)   chkNav.checked=getBool(K_NAV,true);
            applyMiniCartHover(chkHover?.checked??true);
            applyNavbar(chkNav?.checked??true);
            chkHover?.addEventListener('change',()=>{setBool(K_HOVER,chkHover.checked);applyMiniCartHover(chkHover.checked);});
            chkNav?.addEventListener('change',()=>{setBool(K_NAV,chkNav.checked);applyNavbar(chkNav.checked);});
            btnReset?.addEventListener('click',()=>{localStorage.removeItem(K_HOVER);localStorage.removeItem(K_NAV); if(chkHover) chkHover.checked=true; if(chkNav) chkNav.checked=true; applyMiniCartHover(true); applyNavbar(true);});
            function applyMiniCartHover(enabled){
                const host=document.getElementById('nav-cart'); if(!host) return;
                const clone=host.cloneNode(true); host.parentNode.replaceChild(clone,host);
                const panel=document.getElementById('mini-cart-panel');
                if(panel && !enabled) panel.classList.add('d-none');
                if(!enabled) return;
                if(window.__isAuthenticated__!==true) return;
                clone.addEventListener('mouseenter',()=>{ if(typeof showMiniCart==="function") showMiniCart(); });
                clone.addEventListener('mouseleave',()=>{ if(typeof hideMiniCart==="function") hideMiniCart(); });
            }
            function applyNavbar(sticky){
                const navbar=document.querySelector(".navbar-custom"); if(!navbar) return;
                if(sticky){ navbar.style.top="0"; window.updateNavbarVisibility=function(){ navbar.style.top="0"; }; }
                else{ window.updateNavbarVisibility=function(mouseY){ if(window.scrollY===0 || (mouseY??999)<=600) navbar.style.top="0"; else navbar.style.top="-100px"; }; }
            }

            // AJAX модал
            const dlg=document.getElementById('ajaxDialog');
            const dlgTitle=document.getElementById('ajaxDialogTitle');
            const dlgBody=document.getElementById('ajaxDialogBody');
            document.querySelectorAll('.js-ajax-link').forEach(a=>{
                a.addEventListener('click', async (e)=>{
                    e.preventDefault();
                    const url=a.getAttribute('href');
                    dlgTitle.textContent=a.getAttribute('data-title')||'Loading…';
                    dlgBody.innerHTML='<div class="small text-muted">Loading…</div>';
                    try{
                        const res=await fetch(url,{credentials:'same-origin',headers:{'X-Requested-With':'XMLHttpRequest','Accept':'text/html'}});
                        if(!res.ok) throw new Error('Failed to load');
                        const html=await res.text();
                        const tmp=document.createElement('div'); tmp.innerHTML=html;
                        const extracted=tmp.querySelector('#account-page .container, .container, main, body')||tmp;
                        dlgBody.innerHTML=extracted.innerHTML||html;
                    }catch(err){
                        dlgBody.innerHTML=`<div class="alert alert-danger">Could not load content. <a href="${url}">Open in new page</a></div>`;
                    }
                    if(typeof dlg.showModal==='function') dlg.showModal(); else dlg.setAttribute('open','');
                });
            });
            dlg?.querySelector('[data-close]')?.addEventListener('click',()=>{ if(typeof dlg.close==='function') dlg.close(); else dlg.removeAttribute('open'); });

            // AJAX logout (ако сложиш бутона тук)
            const logoutForm=document.getElementById('logoutForm');
            if(logoutForm){
                logoutForm.addEventListener('submit',async (e)=>{
                    e.preventDefault();
                    const fd=new FormData(logoutForm);
                    const res=await fetch('/Account/Logout',{method:'POST',body:fd,credentials:'same-origin',headers:{'X-Requested-With':'XMLHttpRequest','Accept':'application/json'}});
                    try{
                        const data=await res.json();
                        if(data.ok){ window.location.href=data.redirect||'/'; } else { window.location.href='/'; }
                    }catch{ window.location.href='/'; }
                });
            }
        })();
    </script>
}
