@section Scripts {
    <script>
        (function () {
            const redeemForm = document.getElementById('redeemForm');
            const redeemInput = document.getElementById('redeemInput');

            function showRedeemModal(kind, title, message) {
                const modalEl = document.getElementById('redeemResultModal');
                const titleText = document.getElementById('redeemModalTitleText');
                const iconEl = document.getElementById('redeemModalIcon');
                const bodyEl = document.getElementById('redeemModalBody');
                modalEl.classList.remove('redeem-success', 'redeem-warning', 'redeem-error');
                if (kind === 'success') modalEl.classList.add('redeem-success');
                else if (kind === 'warning') modalEl.classList.add('redeem-warning');
                else modalEl.classList.add('redeem-error');
                titleText.textContent = title; bodyEl.textContent = message;
                iconEl.textContent = kind === 'success' ? '✓' : (kind === 'warning' ? '!' : '×');
                new bootstrap.Modal(modalEl).show();
            }

            if (redeemForm) {
                redeemForm.addEventListener('submit', async function (e) {
                    e.preventDefault();
                    const code = (redeemInput?.value || '').trim();
                    if (!code) return;
                    const fd = new FormData(redeemForm);
                    try {
                        const resp = await fetch('@redeemEndpoint', { method: 'POST', body: fd, credentials: 'same-origin', headers: { 'X-Requested-With': 'XMLHttpRequest' } });
                        const ct = resp.headers.get('content-type') || '';
                        const res = ct.includes('application/json') ? await resp.json() : { ok: resp.ok, reason: resp.ok ? 'success' : 'invalid' };
                        if (res.ok === true) { showRedeemModal('success', 'Code Applied', 'Your code has been successfully applied.'); redeemInput.value = ''; return; }
                        if (res.reason === 'already-used' || res.reason === 'already') { showRedeemModal('warning', 'Code Already Used', 'This code has already been redeemed on your account.'); return; }
                        if (res.reason === 'invalid' || res.reason === 'empty' || res.reason === 'expired' || res.reason === 'disabled') { showRedeemModal('error', 'Invalid Code', 'The code is invalid, expired or inactive.'); return; }
                        showRedeemModal('error', 'Something Went Wrong', 'Please try again in a moment.');
                    } catch { showRedeemModal('error', 'Network Error', 'Could not contact the server. Please try again.'); }
                });
            }

            const token = document.querySelector('input[name="__RequestVerificationToken"]')?.value;
            const headers = token ? { 'X-Requested-With': 'XMLHttpRequest', 'RequestVerificationToken': token } : { 'X-Requested-With': 'XMLHttpRequest' };
            const dz = document.getElementById('dropzone');
            const fi = document.getElementById('fileInput');
            const status = document.getElementById('uploadStatus');
            const grid = document.getElementById('uploadsGrid');

            function setStatus(msg, ok) { status.textContent = msg || ''; status.classList.toggle('text-success', !!ok); status.classList.toggle('text-danger', ok === false); }

            async function upload(file) {
                if (!file) return; setStatus('Uploading...');
                const fd = new FormData(); fd.append('file', file); if (token) fd.append('__RequestVerificationToken', token);
                try {
                    const r = await fetch('@uploadEndpoint', { method: 'POST', body: fd, credentials: 'same-origin', headers: { 'X-Requested-With': 'XMLHttpRequest' } });
                    const d = await r.json(); if (!d.ok) throw new Error(d.msg || 'Upload failed'); setStatus('');
                    const col = document.createElement('div'); col.className = 'col';
                    col.innerHTML = `<div class="upload-item">
                        <button type="button" class="btn p-0 border-0 pick-upload" data-path="${d.path}" style="background:none;">
                          <img src="${d.path}" class="rounded border" alt="uploaded" />
                        </button>
                        <button type="button" class="upload-remove" data-path="${d.path}" title="Remove">&times;</button>
                      </div>`;
                    grid.classList.remove('d-none'); grid.prepend(col); bindUploadItem(col);
                } catch (e) { setStatus(e.message, false); }
            }

            dz?.addEventListener('dragover', e => { e.preventDefault(); dz.classList.add('ph-dropzone--over'); });
            dz?.addEventListener('dragleave', () => dz.classList.remove('ph-dropzone--over'));
            dz?.addEventListener('drop', e => { e.preventDefault(); dz.classList.remove('ph-dropzone--over'); upload(e.dataTransfer.files?.[0]); });
            fi?.addEventListener('change', () => upload(fi.files?.[0]));

            function bindUploadItem(root) {
                root.querySelector('.pick-upload')?.addEventListener('click', async (e) => {
                    const path = e.currentTarget.getAttribute('data-path');
                    const fd = new FormData(); fd.append('path', path); if (token) fd.append('__RequestVerificationToken', token);
                    await fetch('@setUploadedEndpoint', { method: 'POST', body: fd, credentials: 'same-origin', headers: { 'X-Requested-With': 'XMLHttpRequest' } });
                });
                root.querySelector('.upload-remove')?.addEventListener('click', async (e) => {
                    const path = e.currentTarget.getAttribute('data-path');
                    const r = await fetch('@deleteUploadEndpoint?path=' + encodeURIComponent(path), { method: 'POST', credentials: 'same-origin', headers });
                    if (r.ok) { root.remove(); }
                });
            }
            document.querySelectorAll('#uploadsGrid .col').forEach(bindUploadItem);
        })();
    </script>
}
